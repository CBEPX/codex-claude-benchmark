class RunnerTests(unittest.TestCase):
    def setUp(self):
        self.assertTrue(callable(getattr(b, 'run_process', None)), 'run_process missing')
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.root = Path(self.tmp.name)

    def run_cmd(self, source, timeout=3):
        return b.run_process([sys.executable, '-c', source], self.root, self.root, timeout)

    def assert_dead(self, pid):
        p = subprocess.run(['ps', '-o', 'stat=', '-p', str(pid)], capture_output=True, text=True)
        self.assertTrue(p.returncode != 0 or p.stdout.strip().startswith('Z'), p.stdout)

    def test_success_and_elapsed(self):
        r = self.run_cmd('import time; time.sleep(.05); print("done")')
        self.assertEqual(r['status'], 'ok')
        self.assertEqual(r['returncode'], 0)
        self.assertGreaterEqual(r['client_seconds'], .05)
        self.assertIn('done', (self.root / 'stdout.jsonl').read_text())

    def test_failure_is_not_success(self):
        r = self.run_cmd('import sys; print("failed", file=sys.stderr); sys.exit(7)')
        self.assertEqual(r['status'], 'error')
        self.assertEqual(r['returncode'], 7)

    def test_timeout_kills_descendant(self):
        source = ('import subprocess,sys,time; '
                  'p=subprocess.Popen([sys.executable,"-c","import time;time.sleep(30)"]); '
                  'open("pid","w").write(str(p.pid)); time.sleep(30)')
        r = self.run_cmd(source, .3)
        self.assertEqual(r['status'], 'timeout')
        self.assertLess(r['client_seconds'], 4)
        self.assert_dead(int((self.root / 'pid').read_text()))

    def test_success_does_not_leave_background_child(self):
        source = ('import subprocess,sys; '
                  'p=subprocess.Popen([sys.executable,"-c","import time;time.sleep(30)"]); '
                  'open("pid","w").write(str(p.pid))')
        self.run_cmd(source)
        self.assert_dead(int((self.root / 'pid').read_text()))

    def test_timeout_kills_detached_descendant(self):
        source = ('import subprocess,sys,time; '
                  'p=subprocess.Popen([sys.executable,"-c","import time;time.sleep(30)"],start_new_session=True); '
                  'open("detached-pid","w").write(str(p.pid)); time.sleep(30)')
        r = self.run_cmd(source, .5)
        pid = int((self.root / 'detached-pid').read_text())
        def cleanup():
            try: os.kill(pid, signal.SIGKILL)
            except ProcessLookupError: pass
        self.addCleanup(cleanup)
        self.assertEqual(r['status'], 'timeout')
        self.assert_dead(pid)


class ContractTests(unittest.TestCase):
    def test_resume_does_not_launch_after_prior_isolation_failure(self):
        with tempfile.TemporaryDirectory() as d:
            root=Path(d);shutil.copytree(b.ROOT/'fixtures/write',root/'fixtures/write')
            row={'client':'claude','mode':'controlled','task':'write','rep':1}
            (root/'manifest.json').write_text(json.dumps({'schedule':[row]}))
            (root/'isolation-preflight.json').write_text(json.dumps({'customization_free':True}))
            (root/'results.jsonl').write_text(json.dumps(dict(row,run_id='prior',status='ok',advisor_calls=1))+'\n')
            with patch.object(b,'ROOT',root),patch.object(b,'check_manifest'),patch.object(b,'run_process',side_effect=RuntimeError('would launch another model')) as run:
                with self.assertRaisesRegex(RuntimeError,'prior isolation'):
                    b.run_series()
                self.assertFalse(run.called)

    def test_total_walltime_includes_event_parsing(self):
        with tempfile.TemporaryDirectory() as d:
            root=Path(d)
            for name in ['benchmark.py','fixtures.py','test_benchmark.py','plan.md']:
                shutil.copy2(b.ROOT/name,root/name)
            for name in ['fixtures','oracle']:
                shutil.copytree(b.ROOT/name,root/name)
            profile=root/'profiles/codex-controlled';profile.mkdir(parents=True)
            shutil.copy2(b.ROOT/'profiles/codex-controlled/config.toml',profile/'config.toml')
            shutil.copy2(b.ROOT/'profiles/claude-clean.json',root/'profiles/claude-clean.json')
            (root/'isolation-preflight.json').write_text(json.dumps({'customization_free':True,'fake':True}))
            fake=root/'fake-bin';fake.mkdir()
            script=f'#!{sys.executable}\nimport pathlib,shutil,json\nw=pathlib.Path.cwd();r=pathlib.Path(__file__).resolve().parents[1]\nshutil.copy2(r/"oracle/write.md",w/"result.md")\nprint(json.dumps({{"type":"turn.completed","usage":{{"output_tokens":42}}}}))\n'
            for name in b.CLIENTS:
                p=fake/name;p.write_text(script);p.chmod(0o700)
            original=b.event_summary
            def slow_summary(*args):
                time.sleep(.2)
                return original(*args)
            with patch.object(b,'ROOT',root),patch.object(b,'event_summary',slow_summary):
                b.freeze();b.run_series(limit=1,fake_bin=fake)
            row=json.loads((root/'results.jsonl').read_text())
            self.assertTrue(row['passed'],row['errors'])
            self.assertGreaterEqual(row['total_seconds']-row['client_seconds']-row['validation_seconds'],.18)
